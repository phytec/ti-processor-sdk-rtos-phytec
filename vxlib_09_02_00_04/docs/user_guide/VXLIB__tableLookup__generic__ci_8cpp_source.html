57 // this method initializes the kernel-specific parameters
58 // mainly, the streaming engine and streaming address generators
59 template <uint32_t dTypeIn, uint32_t dTypeOut>
60 VXLIB_STATUS VXLIB_tableLookup_generic_init_ci(VXLIB_kernelHandle handle,
61  const VXLIB_bufParams2D_t *bufParamsIn,
62  const VXLIB_bufParams2D_t *bufParamsOut,
63  const VXLIB_bufParams1D_t *bufParamsLut,
64  const VXLIB_tableLookup_InitArgs *pKerInitArgs)
65 {
66 #if VXLIB_DEBUGPRINT
67  printf("Enter VXLIB_tableLookup_generic_init_ci\n");
68 #endif
69  VXLIB_STATUS status = VXLIB_SUCCESS; // assign status to success by default
70  VXLIB_tableLookup_PrivArgs *pKerPrivArgs = (VXLIB_tableLookup_PrivArgs *) handle;
71 
72  // structs to hold SE and SA parameters
73  __SE_TEMPLATE_v1 se0Params = __gen_SE_TEMPLATE_v1();
74  __SA_TEMPLATE_v1 sa0Params = __gen_SA_TEMPLATE_v1();
75  size_t width = pKerPrivArgs->width;
76  size_t height = pKerPrivArgs->height;
77  // typecast handle (void) to struct pointer type associated to kernel
78 
79  uint8_t *pBlock = pKerPrivArgs->bufPblock; // address to retrieve to store SE/SA params
80 
81  // set SE0, and SA0 params
82  VXLIB_SE0SA0_init<dTypeIn, dTypeIn>(&se0Params, &sa0Params, bufParamsIn, bufParamsOut);
83  sa0Params.VECLEN = __SA_VECLEN_1ELEM;
84  se0Params.VECLEN = __SE_VECLEN_1ELEM;
85 
86  pKerPrivArgs->numBlocksLut = (height * width) / 8;
87  /**************************/
88  /* Store SE and SA params */
89  /**************************/
90  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET) = se0Params;
91  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET) = sa0Params;
92 #if VXLIB_DEBUGPRINT
93  printf("Exit VXLIB_tableLookup_generic_init_ci\n");
94 #endif
95  return status;
96 }

174  // typecast handle (void) to struct pointer type associated to kernel
175  VXLIB_tableLookup_PrivArgs *pKerPrivArgs = (VXLIB_tableLookup_PrivArgs *) handle;
176 
177  // obtain image parameters
178  size_t width = pKerPrivArgs->width;
179  size_t height = pKerPrivArgs->height;
180  dataType offset = pKerPrivArgs->pKerInitArgs.offset;
181 
182  typedef typename c7x::make_full_vector<dataType>::type vec;
183 
184  /******************************* INDEX FINDING ******************************/
185  uint8_t *pBlock = pKerPrivArgs->bufPblock;
186  int32_t numBlocksLut = pKerPrivArgs->numBlocksLut;
187  __SE_TEMPLATE_v1 se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET);
188  __SA_TEMPLATE_v1 sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET);
189 
190  dataType *restrict pInLocal = (dataType *) pIn;
191  dataType *restrict pOutLocal = (dataType *) pOut;
192  dataType *restrict lutLocal = (dataType *) pLut;

194  printf("In VXLIB_tableLookup_generic_exec_ci, width: %d, height: %d\n", width, height);
195 #endif
196 
197  __SE0_OPEN(pInLocal, se0Params);
198  __SA0_OPEN(sa0Params);
199 
200  int32_t remElem = (height * width) - (numBlocksLut * 8);
201  for (int32_t j = 0; j < numBlocksLut; j++) {
202  vec srcVec = c7x::strm_eng<0, vec>::get_adv();

204  dataType lutValue = lutLocal[srcIndex];
