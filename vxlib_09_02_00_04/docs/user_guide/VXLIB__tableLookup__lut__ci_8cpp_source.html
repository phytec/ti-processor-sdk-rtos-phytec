86 template <uint32_t dTypeIn, uint32_t dTypeOut>
87 VXLIB_STATUS VXLIB_tableLookup_lut_init_ci(VXLIB_kernelHandle handle,
88  const VXLIB_bufParams2D_t *bufParamsIn,
89  const VXLIB_bufParams2D_t *bufParamsOut,
90  const VXLIB_bufParams1D_t *bufParamsLut,
91  const VXLIB_tableLookup_InitArgs *pKerInitArgs)
92 {
93  VXLIB_STATUS status = VXLIB_SUCCESS; // assign status to success by default
94 
95  // structs to hold SE and SA parameters
96 
97  // typecast handle (void) to struct pointer type associated to kernel
98  VXLIB_tableLookup_PrivArgs *pKerPrivArgs = (VXLIB_tableLookup_PrivArgs *) handle;
99 
100  // obtain image size and compute number of blocks to process
101  size_t width = pKerPrivArgs->width;
102  size_t height = pKerPrivArgs->height;
103  size_t elemCount = VXLIB_max_simd<dTypeIn>::value;
104  size_t promotElemCount = VXLIB_max_simd<VXLIB_UINT32>::value;
105  int32_t strideIn = pKerPrivArgs->strideInElements;
106  int32_t strideOut = pKerPrivArgs->strideOutElements;
107 
108  typedef typename VXLIB_vec_type<dTypeIn>::type vec;
109 
110  uint8_t *pBlock = pKerPrivArgs->bufPblock; // address to retrieve to store SE/SA params
111 
112  // set SE0, and SA0 params
113  int32_t nTiles = (width + elemCount - 1) / elemCount;
114  int32_t widthBlock = getWidthBlockLut<dTypeIn>(nTiles);
115 
116  __SE_TEMPLATE_v1 se0Params = __gen_SE_TEMPLATE_v1();
117  __SA_TEMPLATE_v1 sa0Params = __gen_SA_TEMPLATE_v1();
118 
119  __SE_VECLEN SE_VECLEN = c7x::se_veclen<c7x::int_vec>::value;
120  __SA_VECLEN SA_VECLEN = c7x::sa_veclen<vec>::value;
121  __SE_ELETYPE SE_ELETYPE = c7x::se_eletype<vec>::value;
122 
123  sa0Params.ICNT0 = elemCount;
124  sa0Params.DIM1 = elemCount;
125  sa0Params.ICNT1 = nTiles;
126  sa0Params.ICNT2 = height;
127  sa0Params.DIM2 = strideOut;
128  sa0Params.DIMFMT = __SA_DIMFMT_3D;
129  sa0Params.VECLEN = SA_VECLEN;
130  sa0Params.DECDIM1 = __SA_DECDIM_DIM1;
131  sa0Params.DECDIM1SD = __SA_DECDIMSD_DIM0;
132  sa0Params.DECDIM1_WIDTH = width;
133 
134  se0Params.ICNT0 = promotElemCount;
135  se0Params.DIM1 = promotElemCount * 2;
136  se0Params.ICNT1 = setSEICNT1<dTypeIn>(nTiles);
137  se0Params.ICNT2 = height;
138  se0Params.DIM2 = strideIn;
139  se0Params.DIMFMT = __SE_DIMFMT_3D;
140  se0Params.VECLEN = SE_VECLEN;
141  se0Params.DECDIM1 = __SE_DECDIM_DIM1;
142  se0Params.DECDIM1SD = __SE_DECDIMSD_DIM0;
143  se0Params.DECDIM1_WIDTH = width;
144  setSEPromote<dTypeIn>(se0Params);
145  se0Params.ELETYPE = SE_ELETYPE;
146 
147  pKerPrivArgs->numBlocksLut = widthBlock * height;
148 
149  /**************************/
150  /* Store SE and SA params */
151  /**************************/
152  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET) = se0Params;
153  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET) = sa0Params;
154 #if VXLIB_DEBUGPRINT
155  printf("Exit VXLIB_tableLookup_lut_init_ci\n");
156 #endif
157  return status;
158 }

224 template <> inline void lookUpTable_init<uint8_t>(void *lutLocal, uint32_t tableSize)
225 {
226  uint64_t *pData = (uint64_t *) lutLocal;
227  c7x::ulong_vec vData;
228  uint32_t rowOffset = 0;
229  uint32_t rowIncrement = 8;
230  uint32_t loopCount = ((tableSize + rowIncrement - 1) / rowIncrement);
231  for (uint32_t j = 0; j < loopCount; j++) {
232  // Read the quantum into a vector of ulong8
233  vData = (c7x::ulong_vec)(pData[j]);
234 
235  // Update the quanta in all N ways
236  __lookup_init(__LUT_SET0, __as_uchar64(vData), rowOffset);
237 
238  // Increment the row_offset by 8 as we are initializing 8 8-bit elements at a time
239  rowOffset += rowIncrement;
240  }
241 }

296 template <typename dataType> VXLIB_STATUS VXLIB_tableLookup_lut_set_ci(VXLIB_kernelHandle handle, void *restrict pLut)
297 {
298 #if VXLIB_DEBUGPRINT
299  printf("Enter VXLIB_tableLookup_lut_set_ci\n");
300 #endif
301  VXLIB_STATUS status = VXLIB_SUCCESS; // assign status to success by default
302  // typecast handle (void) to struct pointer type associated to kernel
303  VXLIB_tableLookup_PrivArgs *pKerPrivArgs = (VXLIB_tableLookup_PrivArgs *) handle;
304  uint32_t count = pKerPrivArgs->pKerInitArgs.count;
305  dataType *restrict lutLocal = (dataType *) pLut;
306  const uint32_t tableSize = count;
307  // Upper Bits LUT - Contains bits 63-32 for all lookup table values
308  __sLTCRFlags_t set0flags;
309 
310  set0flags.INTERPOLATION = __LUT_INTERP_OFF;
311  set0flags.SATURATION = __LUT_SAT_OFF;
312  set0flags.NUMTBLS = __LUT_NUM_TBL_16;
313  set0flags.TBLSIZE = __LUT_TBL_SIZE_16KBYTES; // Total table size
314  set0flags.WEIGHTSIZE = __LUT_WSIZE_8BIT;
315 
316  setLTCRFlag<dataType>(set0flags);
317 
318  __LUT_SET_LTCR(__LUT_SET0, __lut_set_param(&set0flags));
319  // Start the table at offset 0 from the beginning of L1D-SRAM
320  __LUT_SET_LTBR(__LUT_SET0, 0x0000);
321  // Enable set 0
322  __LUT_SET_LTER(__LUT_ENABLE_0);
323 
324  lookUpTable_init<dataType>(lutLocal, tableSize);
325 #if VXLIB_DEBUGPRINT
326  printf("Exit VXLIB_tableLookup_lut_set_ci\n");
327 #endif
328  return (status);
329 }
