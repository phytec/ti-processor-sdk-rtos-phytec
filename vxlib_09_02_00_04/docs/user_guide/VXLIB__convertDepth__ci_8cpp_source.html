66 // this method initializes the kernel-specific parameters
67 // mainly, the streaming engine and streaming address generators
68 template <uint32_t dTypeIn0, uint32_t dTypeOut>
69 VXLIB_STATUS VXLIB_convertDepth_init_ci(VXLIB_kernelHandle handle,
70  const VXLIB_bufParams2D_t *bufParamsIn0,
71  const VXLIB_bufParams2D_t *bufParamsOut,
72  const VXLIB_convertDepth_InitArgs *pKerInitArgs)
73 {
74  VXLIB_STATUS status = VXLIB_SUCCESS;
75  __SE_TEMPLATE_v1 se0Params;
76  __SE_TEMPLATE_v1 se1Params;
77  __SA_TEMPLATE_v1 sa0Params;
78 
79  __SE_ELETYPE SE_ELETYPE;
80  __SE_VECLEN SE_VECLEN;
81  __SA_VECLEN SA_VECLEN;
82 
83  VXLIB_convertDepth_PrivArgs *pKerPrivArgs = (VXLIB_convertDepth_PrivArgs *) handle;
84  uint8_t *pBlock = pKerPrivArgs->bufPblock;
85 
86  /**********************************************************************/
87  /* Prepare SE/SA params for 8->16 bit images */
88  /**********************************************************************/
89 
90  if ((dTypeIn0 == VXLIB_UINT8) && (dTypeOut == VXLIB_INT16)) {
91  SE_VECLEN = c7x::se_veclen<c7x::short_vec>::value;
92  SE_ELETYPE = c7x::se_eletype<c7x::uchar_vec>::value;
93  SA_VECLEN = c7x::sa_veclen<c7x::short_vec>::value;
94 
95  /**********************************************************************/
96  /* SE/SA params for stride != width 8->16 bit image */
97  /**********************************************************************/
98 
99  // SE parameters are the same regardless if image width is > or < SIMD_WIDTH
100  if (pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_NARROW ||
101  pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_WIDE) {
102 
103  size_t elemCount = VXLIB_max_simd<dTypeIn0>::value;
104  size_t wBlocks = (pKerPrivArgs->width + (elemCount - 1)) / (elemCount);
105  size_t numBlocks = pKerPrivArgs->height * wBlocks;
106 
107  if (pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_WIDE) {
108  pKerPrivArgs->numBlocks = numBlocks * 2;
109  }
110  else {
111  pKerPrivArgs->numBlocks = numBlocks;
112  }
113 
114  /**********************************************************************/
115  /* Prepare SE0 template to fetch A matrix */
116  /**********************************************************************/
117 
118  se0Params = __gen_SE_TEMPLATE_v1();
119  se0Params.ICNT0 = bufParamsIn0->dim_x;
120  se0Params.ICNT1 = bufParamsIn0->dim_y;
121  se0Params.DIMFMT = __SE_DIMFMT_2D;
122  se0Params.PROMOTE = __SE_PROMOTE_2X_ZEROEXT;
123 
124  se0Params.DIM1 = pKerPrivArgs->strideIn0Elements;
125  se0Params.ELETYPE = SE_ELETYPE;
126  se0Params.VECLEN = SE_VECLEN;
127 
128  /**********************************************************************/
129  /* Prepare SA0 template to fetch C matrix */
130  /**********************************************************************/
131  sa0Params = __gen_SA_TEMPLATE_v1();
132  sa0Params.ICNT0 = bufParamsOut->dim_x;
133  sa0Params.ICNT1 = bufParamsOut->dim_y;
134  sa0Params.DIMFMT = __SA_DIMFMT_2D;
135  sa0Params.DIM1 = pKerPrivArgs->strideOutElements;
136  sa0Params.VECLEN = SA_VECLEN;
137 
138  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
139  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
140  }
141 
142  /**********************************************************************/
143  /* SE/SA params for stride = width 8->16 bit image */
144  /**********************************************************************/
145 
146  else if (pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_1D) {
147  /**********************************************************************/
148  /* Prepare SE0 template to fetch A matrix */
149  /**********************************************************************/
150  size_t elemCount = VXLIB_max_simd<dTypeIn0>::value;
151  size_t numBlocks = (bufParamsIn0->dim_x * bufParamsIn0->dim_y) / (elemCount / 2);
152  if (((bufParamsIn0->dim_x * bufParamsIn0->dim_y) % (elemCount / 2)) != 0) {
153  numBlocks++;
154  }
155  pKerPrivArgs->numBlocks = numBlocks;
156 
157  se0Params = __gen_SE_TEMPLATE_v1();
158  se0Params.ICNT0 = bufParamsIn0->dim_x * bufParamsIn0->dim_y;
159  se0Params.DIMFMT = __SE_DIMFMT_1D;
160  se0Params.PROMOTE = __SE_PROMOTE_2X_ZEROEXT;
161  se0Params.ELETYPE = SE_ELETYPE;
162  se0Params.VECLEN = SE_VECLEN;
163 
164  /**********************************************************************/
165  /* Prepare SA0 template to fetch C matrix */
166  /**********************************************************************/
167  sa0Params = __gen_SA_TEMPLATE_v1();
168  sa0Params.ICNT0 = bufParamsOut->dim_x * bufParamsOut->dim_y;
169  sa0Params.DIMFMT = __SA_DIMFMT_1D;
170  sa0Params.VECLEN = SA_VECLEN;
171 
172  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
173  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
174  }
175  else {
176  /* Do nothing */
177  }
178  }
179 
180  /**********************************************************************/
181  /* Prepare SE/SA params for 16->8 bit images */
182  /**********************************************************************/
183  else {
184  SE_VECLEN = c7x::se_veclen<c7x::short_vec>::value;
185  SE_ELETYPE = c7x::se_eletype<c7x::short_vec>::value;
186  SA_VECLEN = c7x::sa_veclen<c7x::uchar_vec>::value;
187 
188  /*******************************************************************************/
189  /* SE/SA params for stride != width and width > SIMD_WIDTH for 16->8 bit image */
190  /*******************************************************************************/
191 
192  if (pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_WIDE) {
193 
194  size_t elemCount = VXLIB_max_simd<dTypeIn0>::value;
195  size_t wBlocks = (pKerPrivArgs->width + 4 * elemCount - 1) / (4 * elemCount);
196  size_t numBlocks = pKerPrivArgs->height * wBlocks;
197 
198  if (pKerPrivArgs->width <= 2 * elemCount) {
199  pKerPrivArgs->numBlocks = (numBlocks + 1) / 2;
200  }
201  else {
202  pKerPrivArgs->numBlocks = numBlocks * 2;
203  }
204 
205  /**********************************************************************/
206  /* Prepare SE0 template to fetch A matrix */
207  /**********************************************************************/
208  se0Params = __gen_SE_TEMPLATE_v1();
209  se0Params.ICNT0 = elemCount;
210  se0Params.ICNT1 = (bufParamsIn0->dim_x + 2 * elemCount - 1) / (2 * elemCount);
211  se0Params.ICNT2 = bufParamsIn0->dim_y;
212  se0Params.DIMFMT = __SE_DIMFMT_3D;
213  se0Params.DIM1 = 2 * elemCount;
214  se0Params.DIM2 = pKerPrivArgs->strideIn0Elements;
215  se0Params.DECDIM1 = __SE_DECDIM_DIM1;
216  se0Params.DECDIM1_WIDTH = bufParamsIn0->dim_x;
217  se0Params.DECDIM2 = __SE_DECDIM_DIM2;
218  se0Params.DECDIM2_WIDTH = bufParamsIn0->stride_y * bufParamsIn0->dim_y;
219  se0Params.ELETYPE = SE_ELETYPE;
220  se0Params.VECLEN = SE_VECLEN;
221 
222  /**********************************************************************/
223  /* Prepare SE1 template to fetch A matrix */
224  /**********************************************************************/
225  se1Params = __gen_SE_TEMPLATE_v1();
226  se1Params.ICNT0 = elemCount;
227  se1Params.ICNT1 = (bufParamsIn0->dim_x + 2 * elemCount - 1) / (2 * elemCount);
228  se1Params.ICNT2 = bufParamsIn0->dim_y;
229  se1Params.DIMFMT = __SE_DIMFMT_3D;
230  se1Params.DIM1 = 2 * elemCount;
231  se1Params.DIM2 = pKerPrivArgs->strideIn0Elements;
232  se1Params.DECDIM1 = __SE_DECDIM_DIM1;
233  se1Params.DECDIM1_WIDTH = bufParamsIn0->dim_x - elemCount;
234  se1Params.DECDIM2 = __SE_DECDIM_DIM2;
235  se1Params.DECDIM2_WIDTH = bufParamsIn0->stride_y * bufParamsIn0->dim_y;
236  se1Params.ELETYPE = SE_ELETYPE;
237  se1Params.VECLEN = SE_VECLEN;
238 
239  /**********************************************************************/
240  /* Prepare SA0 template to fetch C matrix */
241  /**********************************************************************/
242  sa0Params = __gen_SA_TEMPLATE_v1();
243  sa0Params.ICNT0 = bufParamsOut->dim_x;
244  sa0Params.ICNT1 = bufParamsOut->dim_y;
245  sa0Params.DIMFMT = __SA_DIMFMT_2D;
246 
247  sa0Params.DIM1 = pKerPrivArgs->strideOutElements;
248  sa0Params.VECLEN = SA_VECLEN;
249 
250  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;
251  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE1_PARAM_OFFSET) = se1Params;
252  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET) = sa0Params;
253  }
254 
255  /***********************************************************/
256  /* SE/SA params for stride = width and for 16->8 bit image */
257  /***********************************************************/
258  else if (pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_1D) {
259 
260  /*******************************************************************************/
261  /* SE params for stride = width and width > SIMD_WIDTH for 16->8 bit image */
262  /*******************************************************************************/
263  size_t elemCount = VXLIB_max_simd<dTypeIn0>::value;
264  if (bufParamsIn0->dim_x > elemCount) {
265 
266  size_t numBlocks = (bufParamsIn0->dim_x * bufParamsIn0->dim_y) / (elemCount * 4);
267  if (((bufParamsIn0->dim_x * bufParamsIn0->dim_y) % (elemCount * 4)) != 0) {
268  numBlocks++;
269  }
270  pKerPrivArgs->numBlocks = numBlocks;
271  uint32_t factor = (bufParamsIn0->dim_x + 2 * elemCount - 1) / (2 * elemCount);
272  /**********************************************************************/
273  /* Prepare SE0 template to fetch A matrix */
274  /**********************************************************************/
275 
276  se0Params = __gen_SE_TEMPLATE_v1();
277  se0Params.ICNT0 = elemCount;
278  se0Params.ICNT1 = bufParamsIn0->dim_y * factor;
279  se0Params.DIMFMT = __SE_DIMFMT_2D;
280  se0Params.DECDIM1 = __SE_DECDIM_DIM1;
281  se0Params.DECDIM1_WIDTH = bufParamsIn0->stride_y * bufParamsIn0->dim_y;
282  se0Params.DIM1 = 2 * elemCount;
283  se0Params.ELETYPE = SE_ELETYPE;
284  se0Params.VECLEN = SE_VECLEN;

287  /* Prepare SE1 template to fetch A matrix */

306  else {
307

311 
312  se0Params = __gen_SE_TEMPLATE_v1();
313  se0Params.ICNT0 = bufParamsIn0->dim_x * bufParamsIn0->dim_y;
314  se0Params.DIMFMT = __SE_DIMFMT_1D;
315  se0Params.ELETYPE = SE_ELETYPE;
316  se0Params.VECLEN = SE_VECLEN;

320  numBlocks++;
321  }

323  }
324

329  // SA params are the same regardless of width < or > SIMD_WIDTH

336  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET) = se0Params;

343  else {

345  size_t wBlocks = (pKerPrivArgs->width + (elemCount - 1)) / (elemCount);
346  size_t numBlocks = pKerPrivArgs->height * wBlocks;
347  pKerPrivArgs->numBlocks = numBlocks;
348  /**********************************************************************/

408 template <> inline void VXLIB_convertDepth_wrap_wide<int16_t, uint8_t>(uint8_t *pOut, size_t size, uint8_t shift)
409 {
410  typedef typename c7x::make_full_vector<int16_t>::type vecIn;
411  typedef typename c7x::make_full_vector<uint8_t>::type vecOut;
412 
413  for (uint32_t counter = 0; counter < size; counter++) {
414  vecIn a0 = c7x::strm_eng<0, vecIn>::get_adv();
415  vecIn b0 = c7x::strm_eng<1, vecIn>::get_adv();
416  vecIn a1 = c7x::strm_eng<0, vecIn>::get_adv();
417  vecIn b1 = c7x::strm_eng<1, vecIn>::get_adv();
418 
419  vecIn a0Shift = __shift_right(a0, c7x::int_vec(shift));
420  vecIn b0Shift = __shift_right(b0, c7x::int_vec(shift));
421  vecIn a1Shift = __shift_right(a1, c7x::int_vec(shift));
422  vecIn b1Shift = __shift_right(b1, c7x::int_vec(shift));
423 
424  vecOut a = c7x::reinterpret<vecOut>(__permute_even_even_char(
425  VX_vperm_data_0_63, c7x::as_uchar_vec(b0Shift), c7x::as_uchar_vec(a0Shift)));
426  vecOut b = c7x::reinterpret<vecOut>(__permute_even_even_char(
427  VX_vperm_data_0_63, c7x::as_uchar_vec(b1Shift), c7x::as_uchar_vec(a1Shift)));
428 
429  __vpred tmp0 = c7x::strm_agen<0, vecOut>::get_vpred();
430  vecOut *addr0 = c7x::strm_agen<0, vecOut>::get_adv(pOut);
431  __vstore_pred(tmp0, addr0, a);
432  __vpred tmp1 = c7x::strm_agen<0, vecOut>::get_vpred();
433  vecOut *addr1 = c7x::strm_agen<0, vecOut>::get_adv(pOut);
434  __vstore_pred(tmp1, addr1, b);
435  }
436 }

438 template <> inline void VXLIB_convertDepth_wrap_wide<uint8_t, int16_t>(int16_t *pOut, size_t size, uint8_t shift)
439 {
440  typedef typename c7x::make_full_vector<int16_t>::type vecOut;
441 
442  for (uint32_t counter = 0; counter < size; counter++) {
443  vecOut a0 = c7x::strm_eng<0, vecOut>::get_adv();
444  vecOut a0Shift = __shift_left(a0, c7x::int_vec(shift));
445 
446  __vpred tmp0 = c7x::strm_agen<0, vecOut>::get_vpred();
447  vecOut *addr0 = c7x::strm_agen<0, vecOut>::get_adv(pOut);
448  __vstore_pred(tmp0, addr0, a0Shift);
449  }

457 template <> inline void VXLIB_convertDepth_sat_wide<int16_t, uint8_t>(uint8_t *pOut, size_t size, uint8_t shift)
458 {
459  typedef typename c7x::make_full_vector<int16_t>::type vecIn;
460  typedef typename c7x::make_full_vector<uint8_t>::type vecOut;
461 
462  vecIn vecZero = (vecIn) 0;
463  vecIn vecUINT8_MAX = (vecIn) UINT8_MAX;
464 
465  for (uint32_t counter = 0; counter < size; counter++) {
466  vecIn a0 = c7x::strm_eng<0, vecIn>::get_adv();
467  vecIn b0 = c7x::strm_eng<1, vecIn>::get_adv();
468  vecIn a1 = c7x::strm_eng<0, vecIn>::get_adv();
469  vecIn b1 = c7x::strm_eng<1, vecIn>::get_adv();
470 
471  vecIn a0Shift = __shift_right(a0, c7x::int_vec(shift));
472  vecIn b0Shift = __shift_right(b0, c7x::int_vec(shift));
473  vecIn a1Shift = __shift_right(a1, c7x::int_vec(shift));
474  vecIn b1Shift = __shift_right(b1, c7x::int_vec(shift));
475 
476  __vpred cmp_max_a0 = __cmp_gt_pred(vecUINT8_MAX, a0Shift);
477  a0Shift = __select(cmp_max_a0, a0Shift, vecUINT8_MAX);
478  __vpred cmp_max_b0 = __cmp_gt_pred(vecUINT8_MAX, b0Shift);
479  b0Shift = __select(cmp_max_b0, b0Shift, vecUINT8_MAX);
480  __vpred cmp_max_a1 = __cmp_gt_pred(vecUINT8_MAX, a1Shift);
481  a1Shift = __select(cmp_max_a1, a1Shift, vecUINT8_MAX);
482  __vpred cmp_max_b1 = __cmp_gt_pred(vecUINT8_MAX, b1Shift);
483  b1Shift = __select(cmp_max_b1, b1Shift, vecUINT8_MAX);
484

677 // this method performs pixel-wise absolute difference of two input images
678 template <typename dTypeIn0, typename dTypeOut>
679 VXLIB_STATUS VXLIB_convertDepth_exec_ci(VXLIB_kernelHandle handle, void *restrict pIn0, void *restrict pOut)
680 {
681  VXLIB_convertDepth_PrivArgs *pKerPrivArgs = (VXLIB_convertDepth_PrivArgs *) handle;
682 
683 #if VXLIB_DEBUGPRINT
684  printf("Enter VXLIB_convertDepth_exec_ci\n");
685 #endif
686 
687  uint8_t *pBlock = pKerPrivArgs->bufPblock;
688  size_t numBlocks = pKerPrivArgs->numBlocks;
689  uint8_t shift = pKerPrivArgs->shift;
690 
691  // int16 is used in both convertDepth types due to SE promote for 8->16 bit
692  size_t SIMD_WIDTH = c7x::max_simd<int16_t>::value;
693  dTypeOut *restrict pOutLocal;
694 
695  /**************************************************************************/
696  /* Open appropriate SE/SA */
697  /**************************************************************************/
698 
699  // For images with widths larger than SIMD_WIDTH pixels and for 16->8 bit conversion, utilize 2 SEs
700  // Ex: For C7100/C7120 if the width of the image is larger than 32 pixels
701  if (pKerPrivArgs->width > SIMD_WIDTH && pKerPrivArgs->data_type == VXLIB_INT16) {
702  __SE_TEMPLATE_v1 se0Params;
703  __SE_TEMPLATE_v1 se1Params;
704  __SA_TEMPLATE_v1 sa0Params;
705 
706  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
707  se1Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE1_PARAM_OFFSET);
708  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
709 
710  dTypeIn0 *restrict pInLocal0 = (dTypeIn0 *) pIn0;
711  dTypeIn0 *restrict pInLocal1 = (dTypeIn0 *) pIn0 + SIMD_WIDTH;
712  pOutLocal = (dTypeOut *) pOut;
713 
714  // Input samples
715  __SE0_OPEN(pInLocal0, se0Params);
716  __SE1_OPEN(pInLocal1, se1Params);
717 
718  // Output samples
719  __SA0_OPEN(sa0Params);
720  }
721  // For images with widths of <= SIMD_WIDTH or for 8->16 bit conversion, utilize 1 SE
722  else {
723  __SE_TEMPLATE_v1 se0Params;
724  __SA_TEMPLATE_v1 sa0Params;
725 
726  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SE0_PARAM_OFFSET);
727  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE_SA0_PARAM_OFFSET);
728 
729  dTypeIn0 *restrict pInLocal0 = (dTypeIn0 *) pIn0;
730  pOutLocal = (dTypeOut *) pOut;
731 
732  // Input samples
733  __SE0_OPEN(pInLocal0, se0Params);
734 
735  // Output samples
736  __SA0_OPEN(sa0Params);
737  }
738 
739  /*************************************************************************/
740  /* Determining calculation loop depending on sePolicy and overflowPolicy */
741  /*************************************************************************/
742 
743  /**************************************************************************/
744  /* Loops for wrap behavior for 16->8 images and all 8->16 images */
745  /**************************************************************************/
746  if (pKerPrivArgs->overFlowPolicy == VXLIB_WRAP && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_NARROW) {
747  VXLIB_convertDepth_wrap_narrow<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
748  }
749  else if (pKerPrivArgs->overFlowPolicy == VXLIB_WRAP && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_WIDE) {
750  VXLIB_convertDepth_wrap_wide<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
751  }
752  else if (pKerPrivArgs->overFlowPolicy == VXLIB_WRAP && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_1D) {
753  if (pKerPrivArgs->width <= SIMD_WIDTH && pKerPrivArgs->data_type == VXLIB_INT16) {
754  VXLIB_convertDepth_wrap_narrow_1D<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
755  }
756  else if (pKerPrivArgs->width <= SIMD_WIDTH && pKerPrivArgs->data_type == VXLIB_UINT8) {
757  VXLIB_convertDepth_wrap_narrow<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
758  }
759  else {
760  VXLIB_convertDepth_wrap_wide<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
761  }
762  }
763 
764  /**************************************************************************/
765  /* Loops for saturate behavior for 16->8 images */
766  /**************************************************************************/
767  else if (pKerPrivArgs->overFlowPolicy == VXLIB_SATURATE && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_1D) {
768  if (pKerPrivArgs->width <= SIMD_WIDTH) {
769  VXLIB_convertDepth_sat_narrow_1D<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
770  }
771  else {
772  VXLIB_convertDepth_sat_wide<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
773  }
774  }
775  else if (pKerPrivArgs->overFlowPolicy == VXLIB_SATURATE && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_WIDE) {
776  VXLIB_convertDepth_sat_wide<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
777  }
778  else if (pKerPrivArgs->overFlowPolicy == VXLIB_SATURATE && pKerPrivArgs->sePolicy == VXLIB_CONVERTDEPTH_SE_NARROW) {
779  VXLIB_convertDepth_sat_narrow<dTypeIn0, dTypeOut>(pOutLocal, numBlocks, shift);
780  }
781  else
782  {
783  /* Nothing to do here.*/
784  }
785 
786  /**************************************************************************/
787  /* Close appropriate SE/SA */
788  /**************************************************************************/
789  if (pKerPrivArgs->width > SIMD_WIDTH && pKerPrivArgs->data_type == VXLIB_INT16) {
790  __SE0_CLOSE();
791  __SE1_CLOSE();
792  __SA0_CLOSE();
793  }
794  else {
795  __SE0_CLOSE();
796  __SA0_CLOSE();
797  }
798 
799  return VXLIB_SUCCESS;
800 }
