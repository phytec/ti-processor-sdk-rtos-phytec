58 void VXLIB_histogram_val_init(uint16_t numBins)
59 {
60  ulong8 vSeed;
61  uint32_t row_offset;
62 
63  row_offset = 0;
64  for (int32_t j = 0; j < numBins; j += 2) {
65  // Let the seed be 0, here one quanta will be 2, 32-bit elements
66  vSeed = (ulong8) (0x0000000000000000);
67 
68  // Update the quanta in all N ways
69  __lookup_init(__LUT_SET0, __as_ushort32(vSeed), row_offset);
70 
71  // Increment the row_offset by 2 as there are 2, 32-bit bins every row in a 64-bit bank
72  row_offset += 2;
73  }
74 }

78 template <uint32_t dTypeIn, uint32_t dTypeOut>
79 VXLIB_STATUS VXLIB_histogram_init_ci(VXLIB_kernelHandle handle,
80  const VXLIB_bufParams2D_t *bufParamsIn,
81  const VXLIB_bufParams1D_t *bufParamsOut,
82  const VXLIB_histogram_InitArgs *pKerInitArgs)
83 {
84  VXLIB_STATUS status = VXLIB_SUCCESS; // assign status to success by default
85 
86  // typecast handle (void) to struct pointer type associated to kernel
87  VXLIB_histogram_PrivArgs *pKerPrivArgs = (VXLIB_histogram_PrivArgs *) handle;
88 
89  // obtain image size and compute number of blocks to process
90  size_t width = pKerPrivArgs->width;
91  size_t height = pKerPrivArgs->height;
92  size_t elemCount = VXLIB_max_simd<dTypeOut>::value;
93  size_t wBlocks = (width + (elemCount - 1)) / (elemCount);
94  size_t numBlocks = height * wBlocks;
95  uint16_t numBins = pKerInitArgs->numBins;
96  size_t strideInElements = pKerPrivArgs->strideInElements;
97  pKerPrivArgs->numBlocks = numBlocks;
98 
99  // structs to hold SE and SA parameters
100  __SE_TEMPLATE_v1 se0Params = __gen_SE_TEMPLATE_v1();
101  __SE_TEMPLATE_v1 se1Params = __gen_SE_TEMPLATE_v1();
102  __SA_TEMPLATE_v1 sa0Params = __gen_SA_TEMPLATE_v1();
103 
104  __SE_ELETYPE SE_ELETYPE;
105  __SE_VECLEN SE_VECLEN;
106  __SA_VECLEN SA_VECLEN;
107 
108  uint8_t *pBlock = pKerPrivArgs->bufPblock; // address to retrieve to store SE/SA params
109 
110  if (VXLIB_HISTOGRAM_I8U_O32U) {
111  SE_ELETYPE = c7x::se_eletype<c7x::uchar_vec>::value;
112  SE_VECLEN = c7x::se_veclen<c7x::int_vec>::value;
113  SA_VECLEN = c7x::sa_veclen<c7x::int_vec>::value;
114 
115  if (width == strideInElements) {
116  se0Params.ELETYPE = SE_ELETYPE;
117  se0Params.VECLEN = SE_VECLEN;
118  se0Params.DIMFMT = __SE_DIMFMT_1D;
119  se0Params.PROMOTE = __SE_PROMOTE_4X_ZEROEXT;
120  se0Params.ICNT0 = width * height;
121  }
122 
123  else {
124  se0Params.ELETYPE = SE_ELETYPE;
125  se0Params.VECLEN = SE_VECLEN;
126  se0Params.DIMFMT = __SE_DIMFMT_2D;
127  se0Params.PROMOTE = __SE_PROMOTE_4X_ZEROEXT;
128 
129  se0Params.ICNT0 = width;
130  se0Params.ICNT1 = height;
131  se0Params.DIM1 = strideInElements;
132  }
133 
134  sa0Params.VECLEN = SA_VECLEN;
135  sa0Params.DIMFMT = __SA_DIMFMT_1D;
136  sa0Params.ICNT0 = numBins;
137  }
138 
139  /**************************/
140  /* Store SE and SA params */
141  /**************************/
142 
143  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE0_PARAM_OFFSET) = se0Params;
144  *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET) = sa0Params;
145 
146  /******************************************/
147  /* Initialize LUT for !C7504 C7x variants */
148  /******************************************/
149 
150  __sLTCRFlags_t set0flags;
151 
152  // 16 way lut for 16-bit data with 64 entries (6-bit index)
153  set0flags.INTERPOLATION = __LUT_INTERP_OFF; // NA, applicable only in LUT mode
154  set0flags.SATURATION = __LUT_SAT_ON; // Saturate after reaching limit
155  set0flags.SIGNUNSIGN = __LUT_UNSIGNED; // data elements are unsigned
156  set0flags.ELETYPE = __LUT_ELETYPE_32BIT; // This is the bit-depth of each bin
157  set0flags.NUMTBLS = __LUT_NUM_TBL_16; // 16 parallel lookup every cycle
158  set0flags.TBLSIZE = __LUT_TBL_SIZE_16KBYTES; // Total table size
159  set0flags.WEIGHTSIZE = __LUT_WSIZE_8BIT; // Set weight size as 8-bit
160  set0flags.PROMOTION = __LUT_PROMOTE_OFF; // NA, applicable only in LUT mode
161 
162  // Set configuration register for SET0
163  __LUT_SET_LTCR(__LUT_SET0, __lut_set_param(&set0flags));
164 
165  // Start the table at offset 0 from the beginning of L1D-SRAM
166  __LUT_SET_LTBR(__LUT_SET0, 0x0000);
167 
168  // Enable set 1
169  __LUT_SET_LTER(__LUT_ENABLE_0);
170 
171  VXLIB_histogram_val_init(numBins);
172 
173  /* Generate predicate buffer */
174  if (VXLIB_HISTOGRAM_I8U_O32U) {
175 
176  if (width != strideInElements) {
177 
178  int32_t blockCounter = 0;
179 
180  uint64_t *predRegister = (uint64_t *) pKerPrivArgs->bufPredicateStore;
181 
182  uint32_t i = 0;
183  int32_t j = 0;
184  uint64_t predictedValue = 0x0;
185 
186  while (blockCounter < wBlocks) {
187  for (i = 0; i < elemCount; i++) {
188  if (j < width) {
189  predictedValue |= (uint64_t) ((uint64_t) 0xF << (i * 4));
190  j++;
191  }
192  else {
193  j++;
194  }
195  }
196  *predRegister = predictedValue;
197 
198  predRegister++;
199  blockCounter++;
200  predictedValue = 0x0;
201  }
202 
203  SE_ELETYPE = c7x::se_eletype<c7x::ulong_vec>::value;
204  SE_VECLEN = c7x::se_veclen<c7x::ulong_vec>::value;
205 
206  se1Params.ELETYPE = __SE_ELETYPE_64BIT;
207  se1Params.VECLEN = __SE_VECLEN_1ELEM;
208  se1Params.DIMFMT = __SE_DIMFMT_3D;
209  se1Params.ICNT0 = 1;
210  se1Params.ICNT1 = wBlocks;
211  se1Params.ICNT2 = height;
212  se1Params.DIM1 = 1;
213  se1Params.DIM2 = 0;
214 
215  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET) = se1Params;
216  }
217 
218  else if (width == strideInElements) {
219 
220  uint64_t *predRegister = (uint64_t *) pKerPrivArgs->bufPredicateStore;
221 
222  uint32_t i = 0;
223  int32_t j = 0;
224  uint64_t predictedValue = 0x0;
225  size_t widthLastBlock = (width * height) % elemCount;
226 
227  for (i = 0; i < elemCount; i++) {
228  if (j < widthLastBlock) {
229  predictedValue |= (uint64_t) ((uint64_t) 0xF << (i * 4));
230  j++;
231  }
232  else {
233  j++;
234  }
235  }
236  *predRegister = predictedValue;
237 
238  se1Params.ELETYPE = __SE_ELETYPE_64BIT;
239  se1Params.VECLEN = __SE_VECLEN_1ELEM;
240  se1Params.DIMFMT = __SE_DIMFMT_1D;
241  se1Params.ICNT0 = 1;
242 
243  *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET) = se1Params;
244  }
245  }
246 
247  return status;
248 }

268 // this method generates a histogram distribution based on pixel values
269 template <typename dTypeIn, typename dTypeOut>
270 VXLIB_STATUS VXLIB_histogram_exec_ci(VXLIB_kernelHandle handle, void *restrict pIn, void *restrict pOut)
271 {
272  // typecast handle (void) to struct pointer type associated to kernel
273  VXLIB_histogram_PrivArgs *pKerPrivArgs = (VXLIB_histogram_PrivArgs *) handle;
274 
275  // structs to retrieve SE and SA paramters
276  __SE_TEMPLATE_v1 se0Params;
277  __SE_TEMPLATE_v1 se1Params;
278  __SA_TEMPLATE_v1 sa0Params;
279 
280  // create local pointers
281  dTypeIn *restrict pInLocal = (dTypeIn *) pIn;
282  dTypeOut *restrict pOutLocal = (dTypeOut *) pOut;
283  uint8_t *restrict pPredicate = (uint8_t *) pKerPrivArgs->bufPredicateStore;
284 
285 #if VXLIB_DEBUGPRINT
286  printf("Enter VXLIB_histogram_exec_ci\n");
287 #endif
288 
289  // address of SE and SA parameters
290  uint8_t *pBlock = pKerPrivArgs->bufPblock;
291 
292  // retrieve SE and SA parameters
293  se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE0_PARAM_OFFSET);
294  se1Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET);
295  sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET);
296 
297  // open SEs to fetch Input samples
298  __SE0_OPEN(pInLocal, se0Params);
299  __SE1_OPEN(pPredicate, se1Params);
300 
301  // open SA0 to write output samples
302  __SA0_OPEN(sa0Params);
303 
304  size_t numBlocks = pKerPrivArgs->numBlocks;
305  uint16_t numBins = pKerPrivArgs->pKerInitArgs.numBins;
306 
307  c7x::uint_vec lowerBound = pKerPrivArgs->pKerInitArgs.offset;
308  c7x::uint_vec upperBound = pKerPrivArgs->pKerInitArgs.range + pKerPrivArgs->pKerInitArgs.offset;
309  c7x::uint_vec numBinsVec = pKerPrivArgs->pKerInitArgs.numBins;
310  c7x::uint_vec rangeVec = pKerPrivArgs->pKerInitArgs.range;
311 
312  // derive c7x vector type from template typename
313  typedef typename c7x::make_full_vector<dTypeOut>::type vec;
314 
315  uint32_t maxSIMD = c7x::max_simd<dTypeOut>::value;
316 
317  /**************************/
318  /* Populate Histogram */
319  /**************************/
320 
321  // Linear images
322  if (pKerPrivArgs->width == pKerPrivArgs->strideInElements) {
323  // When width * height is not a multiple of SIMD, need to use predicate buffer on last block to mask DECDIM 0s
324  if ((pKerPrivArgs->width * pKerPrivArgs->height) % maxSIMD) {
325  size_t linearBlocks = (pKerPrivArgs->width * pKerPrivArgs->height) / maxSIMD;
326  for (uint32_t counter = 0; counter < linearBlocks; counter++) {
327  vec inVec = c7x::strm_eng<0, vec>::get_adv(); // fetch input 0 via SE0
328 
329  __vpred cmp_lower = __cmp_le_pred(lowerBound, inVec);
330  __vpred cmp_upper = __cmp_lt_pred(inVec, upperBound);
331  __vpred condHist = __and(cmp_lower, cmp_upper);
332 
333  vec histIndex = ((inVec - lowerBound) * numBinsVec) / rangeVec;
334 
335  vec inHist = __select(condHist, histIndex, numBinsVec);
336 
337  __hist(__LUT_SET0, inHist);
338  }
339  // Predicate buffer to mask DECDIM 0s on last block
340  vec inVec = c7x::strm_eng<0, vec>::get_adv(); // fetch input 0 via SE0
341  __vpred inPred = _mvrp(c7x::strm_eng<1, ulong>::get_adv());
342 
343  __vpred cmp_lower = __cmp_le_pred(lowerBound, inVec);
344  __vpred cmp_upper = __cmp_lt_pred(inVec, upperBound);
345  __vpred condHist = __and(cmp_lower, cmp_upper);
346  condHist = __and(condHist, inPred);
347 
348  vec histIndex = ((inVec - lowerBound) * numBinsVec) / rangeVec;
349 
350  vec inHist = __select(condHist, histIndex, numBinsVec);
351 
352  __hist(__LUT_SET0, inHist);
353  }
354  // Width * height is a SIMD multiple, can process entire image in one loop
355  else {
356  size_t linearBlocks = (pKerPrivArgs->width * pKerPrivArgs->height) / maxSIMD;
357  for (uint32_t counter = 0; counter < linearBlocks; counter++) {
358  vec inVec = c7x::strm_eng<0, vec>::get_adv(); // fetch input 0 via SE0
359  __vpred cmp_lower = __cmp_le_pred(lowerBound, inVec);
360  __vpred cmp_upper = __cmp_lt_pred(inVec, upperBound);
361  __vpred condHist = __and(cmp_lower, cmp_upper);
362 
363  vec histIndex = ((inVec - lowerBound) * numBinsVec) / rangeVec;
364 
365  vec inHist = __select(condHist, histIndex, numBinsVec);
366 
367  __hist(__LUT_SET0, inHist);
368  }
369  }
370  }
371  // Not linear image (width != stride)
372  else {
373  for (uint32_t counter = 0; counter < numBlocks; counter++) {
374  vec inVec = c7x::strm_eng<0, vec>::get_adv(); // fetch input 0 via SE0
375  __vpred inPred = _mvrp(c7x::strm_eng<1, ulong>::get_adv()); // Predicate buffer to mask DECDIM 0s on last block
376 
377  __vpred cmp_lower = __cmp_le_pred(lowerBound, inVec);
378  __vpred cmp_upper = __cmp_lt_pred(inVec, upperBound);
379  __vpred condHist = __and(cmp_lower, cmp_upper);
380  condHist = __and(condHist, inPred);
381 
382  vec histIndex = ((inVec - lowerBound) * numBinsVec) / rangeVec;
383 
384  // numBinsVec is chosen if the pixel does not meet the criteria to be in the distribution
385  // bins are from bin[0] - bin[numBins -1] and thus numBins will not be a valid index and will not populate
386  // histogram
387  vec inHist = __select(condHist, histIndex, numBinsVec);
388 
389  __hist(__LUT_SET0, inHist);
390  }
391  }
392 
393  /**************************/
394  /* Store Histogram */
395  /**************************/
396 
397  // Store histogram in outBuffer if last call to exec function for image
398  if (pKerPrivArgs->pKerInitArgs.lastBlock) {
399 
400  uint32_t saStoreIter = (numBins / maxSIMD);
401  if (numBins % maxSIMD) {
402  saStoreIter++;
403  }
404  c7x::uint_vec bin;
405  uint16_t binCount = 0;
406 
407  for (int i = 0; i < saStoreIter; i++) {
408  bin = 0;
409  for (int j = 0; j < maxSIMD; j++) {
410  c7x::uint_vec vHist = __lookup_read_uint(__LUT_SET0, (c7x::uint_vec)(binCount));
411  // Horizontal add all bins to one index
412  bin.s[j] = __horizontal_add(vHist);
413  binCount++;
414  }
415 
416  // After reaching a SIMD number of bins, store vector
417  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
418  vec *addr = c7x::strm_agen<0, vec>::get_adv(pOutLocal);
419  __vstore_pred(tmp, addr, bin);
420  }
421  }
422 
423  // close SE/SA
424  __SE0_CLOSE();
425  __SE1_CLOSE();
426  __SA0_CLOSE();
427 
428  return VXLIB_SUCCESS;
429 }
