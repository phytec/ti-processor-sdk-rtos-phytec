43 VXLIB_absDiff_init_checkParams(VXLIB_kernelHandle handle,
44  const VXLIB_bufParams2D_t *bufParamsIn0,
45  const VXLIB_bufParams2D_t *bufParamsIn1,
46  const VXLIB_bufParams2D_t *bufParamsOut,
47  const VXLIB_absDiff_InitArgs *pKerInitArgs)
48 {
49  VXLIB_STATUS status = VXLIB_SUCCESS;
50 
51 #if VXLIB_DEBUGPRINT
52  printf("Enter VXLIB_absDiff_init_checkParams\n");
53 #endif
54  if (handle == NULL || bufParamsIn0 == NULL || bufParamsIn1 == NULL || bufParamsOut == NULL || pKerInitArgs == NULL) {
55  status = VXLIB_ERR_NULL_POINTER;
56  }
57 
58  if (status == VXLIB_SUCCESS) {
59  if (((bufParamsIn0->data_type != VXLIB_INT16) && (bufParamsIn1->data_type != VXLIB_INT16)) &&
60  ((bufParamsIn0->data_type != VXLIB_UINT8) && (bufParamsIn1->data_type != VXLIB_UINT8))) {
61  status = VXLIB_ERR_INVALID_TYPE;
62  }
63  else if (bufParamsIn0->data_type != bufParamsOut->data_type ||
64  bufParamsIn1->data_type != bufParamsOut->data_type) {
65  status = VXLIB_ERR_INVALID_TYPE;
66  }
67  else {
68  /* Nothing to do here */
69  }
70  }
71 
72  if (status == VXLIB_SUCCESS) {
73 
74  uint32_t strideIn0Elements = bufParamsIn0->stride_y / VXLIB_sizeof(bufParamsIn0->data_type);
75  uint32_t strideIn1Elements = bufParamsIn1->stride_y / VXLIB_sizeof(bufParamsIn1->data_type);
76  uint32_t strideOutElements = bufParamsOut->stride_y / VXLIB_sizeof(bufParamsOut->data_type);
77 
78  if ((bufParamsIn0->dim_x != bufParamsIn1->dim_x) || (bufParamsIn0->dim_y != bufParamsIn1->dim_y) ||
79  (bufParamsIn0->dim_x != bufParamsOut->dim_x) || (bufParamsIn0->dim_y != bufParamsOut->dim_y) ||
80  (strideIn0Elements < bufParamsIn0->dim_x) || (strideIn1Elements < bufParamsIn1->dim_x) ||
81  (strideOutElements < bufParamsOut->dim_x)) {
82  status = VXLIB_ERR_INVALID_DIMENSION;
83  }
84  }
85 
86  return status;
87 }

109 VXLIB_STATUS VXLIB_absDiff_init(VXLIB_kernelHandle handle,
110  VXLIB_bufParams2D_t *bufParamsIn0,
111  VXLIB_bufParams2D_t *bufParamsIn1,
112  VXLIB_bufParams2D_t *bufParamsOut,
113  const VXLIB_absDiff_InitArgs *pKerInitArgs)
114 {
115  VXLIB_STATUS status = VXLIB_SUCCESS;
116  VXLIB_absDiff_PrivArgs *pKerPrivArgs = (VXLIB_absDiff_PrivArgs *) handle;
117 
118  uint32_t height = bufParamsOut->dim_y;
119  uint32_t width = bufParamsOut->dim_x;
120 
121  uint32_t datatype = bufParamsIn0->data_type;
122  uint32_t strideIn0Elements = bufParamsIn0->stride_y / VXLIB_sizeof(datatype);
123  uint32_t strideIn1Elements = bufParamsIn1->stride_y / VXLIB_sizeof(datatype);
124  uint32_t strideOutElements = bufParamsOut->stride_y / VXLIB_sizeof(datatype);
125 
126  pKerPrivArgs->height = height;
127  pKerPrivArgs->width = width;
128  pKerPrivArgs->strideIn0Elements = strideIn0Elements;
129  pKerPrivArgs->strideIn1Elements = strideIn1Elements;
130  pKerPrivArgs->strideOutElements = strideOutElements;
131 
132 #if VXLIB_DEBUGPRINT
133  printf("VXLIB_DEBUGPRINT Enter VXLIB_absDiff_init\n");
134 #endif
135 
136  // determine natural C vs optimized function call
137  if (pKerInitArgs->funcStyle == VXLIB_FUNCTION_NATC) {
138 
139  // set function pointer for natural C function with appropriate template parameters based on datatypes
140  if (bufParamsIn0->data_type == VXLIB_UINT8) {
141  pKerPrivArgs->execute = VXLIB_absDiff_exec_cn<uint8_t>;
142  }
143  else if (bufParamsIn0->data_type == VXLIB_INT16) {
144  pKerPrivArgs->execute = VXLIB_absDiff_exec_cn<int16_t>;
145  }
146  else {
147  status = VXLIB_ERR_INVALID_TYPE;
148  }
149  }
150  else {
151 
152  // optimized function
153  // set function pointer for optimized function with appropriate template parameters based on datatypes
154  if (bufParamsIn0->data_type == VXLIB_UINT8) {
155  pKerPrivArgs->execute = VXLIB_absDiff_exec_ci<uint8_t>;
156  status = VXLIB_absDiff_init_ci<VXLIB_UINT8>(handle, bufParamsIn0, bufParamsIn1, bufParamsOut, pKerInitArgs);
157  }
158  else if (bufParamsIn0->data_type == VXLIB_INT16) {
159  pKerPrivArgs->execute = VXLIB_absDiff_exec_ci<int16_t>;
160  status = VXLIB_absDiff_init_ci<VXLIB_INT16>(handle, bufParamsIn0, bufParamsIn1, bufParamsOut, pKerInitArgs);
161  }
162  else {
163  status = VXLIB_ERR_INVALID_TYPE;
164  }
165  }
166 
167  return status;
168 }
