48 // this method computes median in via natural C code
49 template <typename dType>
50 VXLIB_STATUS VXLIB_median_exec_cn(VXLIB_kernelHandle handle,
51  void *restrict pIn,
52  void *restrict pMask,
53  void *restrict pOut,
54  void *restrict pScratch)
55 {
56 
57  VXLIB_STATUS status = VXLIB_SUCCESS; // assign status to success by default
58 
59 #if VXLIB_DEBUGPRINT
60  printf("Enter VXLIB_median_exec_cn\n");
61 #endif
62 
63  // typecast handle (void) to struct pointer type associated to kernel
64  VXLIB_median_PrivArgs *pKerPrivArgs = (VXLIB_median_PrivArgs *) handle;
65 
66  // obtain image parameters and overflow policy
67  size_t widthOut = pKerPrivArgs->widthOut;
68  size_t heightOut = pKerPrivArgs->heightOut;
69  bool kernelType = pKerPrivArgs->pKerInitArgs.kernelType;
70  size_t M = pKerPrivArgs->M;
71  size_t N = pKerPrivArgs->N;
72 
73  size_t strideInElements = pKerPrivArgs->strideInElements;
74  size_t strideMaskElements = pKerPrivArgs->strideMaskElements;
75  size_t strideOutElements = pKerPrivArgs->strideOutElements;
76 
77  // create local pointers
78  dType *restrict pInLocal = (dType *) pIn;
79  dType *restrict pMaskLocal = (dType *) pMask;
80  dType *restrict pOutLocal = (dType *) pOut;
81 
82 #if VXLIB_DEBUGPRINT
83  printf("In VXLIB_median_exec_cn, widthOut: %ld, heightOut: %ld\n", widthOut, heightOut);
84 #endif
85 
86  if (!kernelType) {
87  size_t x, y, j, k;
88  dType inWindow[9];
89  for (y = 0; y < heightOut; y++) {
90  for (x = 0; x < widthOut; x++) {
91 
92  /* Extract 3x3 window into stack array */
93  for (j = 0; j < 3; j++) {
94  for (k = 0; k < 3; k++) {
95  inWindow[j * 3 + k] = pInLocal[(y + j) * strideInElements + (x + k)];
96  }
97  }
98 
99  /* Sort the values in window from largest to smallest */
100  for (j = 0; j < 8; j++) {
101  for (k = 0; k < 8; k++) {
102  if (inWindow[k] < inWindow[k + 1]) {
103  dType tmp = inWindow[k];
104  inWindow[k] = inWindow[k + 1];
105  inWindow[k + 1] = tmp;
106  }
107  }
108  }
109 
110  /* Median is the middle value */
111  pOutLocal[y * strideOutElements + x] = inWindow[4];
112  }
113  }
114  }
115  else {
116  size_t x, y, j, k, cnt;
117  dType inWindow[9 * 9];
118  for (y = 0; y < heightOut; y++) {
119  for (x = 0; x < widthOut; x++) {
120 
121  cnt = 0;
122 
123  /* Extract MxN window into stack array */
124  for (j = 0; j < M; j++) {
125  for (k = 0; k < N; k++) {
126  if (pMaskLocal[j * strideMaskElements + k]) {
127  inWindow[cnt] = pInLocal[(y + j) * strideInElements + (x + k)];
128  cnt++;
129  }
130  }
131  }
132 
133  if (cnt > 0) {
134  /* Sort the values in window from largest to smallest */
135  for (j = 0; j < (cnt - 1); j++) {
136  for (k = 0; k < (cnt - 1); k++) {
137  if (inWindow[k] < inWindow[k + 1]) {
138  dType tmp = inWindow[k];
139  inWindow[k] = inWindow[k + 1];
140  inWindow[k + 1] = tmp;
141  }
142  }
143  }
144 
145  /* Median is the middle value */
146  pOutLocal[y * strideOutElements + x] = inWindow[cnt / 2];
147  }
148  else {
149  status = VXLIB_ERR_FAILURE;
150  VXLIB_DEBUGPRINTFN(0, "ERROR CN: True Count = %lu. Cannot execute kernel\n", cnt);
151  }
152  }
153  }
154  }
155 
156  return (status);
157 }
