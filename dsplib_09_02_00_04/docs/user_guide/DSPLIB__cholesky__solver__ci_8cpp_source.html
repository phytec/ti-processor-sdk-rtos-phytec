229 template <typename dataType>
230 static inline void
231 DSPLIB_cholesky_solver_matTrans_ci(dataType *pIn, dataType *pOut, int32_t order, int32_t colLStride, uint8_t *pBlock)
232 {
233  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering function");
234 
235 
236  __SE_TEMPLATE_v1 se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + (0 * SE_PARAM_SIZE));
237  __SA_TEMPLATE_v1 sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + (1 * SE_PARAM_SIZE));
238 
239  typedef typename c7x::make_full_vector<dataType>::type vec;
240  int32_t eleCount = c7x::element_count_of<vec>::value;
241  int32_t nVec = DSPLIB_ceilingDiv(order, (eleCount * 2));
242 
243  int32_t loopCount = nVec * order;
244  __SE0_OPEN(pIn, se0Params);
245  __SE1_OPEN(pIn + (colLStride * eleCount), se0Params);
246  __SA0_OPEN(sa0Params);
247 
248  for (int32_t i = 0; i < loopCount; i++) {
249  vec v1 = c7x::strm_eng<0, vec>::get_adv();
250  vec v2 = c7x::strm_eng<1, vec>::get_adv();
251 
252  __vpred pred1 = c7x::strm_agen<0, vec>::get_vpred();
253  vec *pStore1 = c7x::strm_agen<0, vec>::get_adv(pOut);
254  __vstore_pred(pred1, pStore1, v1);
255 
256  __vpred pred2 = c7x::strm_agen<0, vec>::get_vpred();
257  vec *pStore2 = c7x::strm_agen<0, vec>::get_adv(pOut);
258  __vstore_pred(pred2, pStore2, v2);
259  }
260 
261  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Exiting function");
262 }

283 template <typename dataType>
284 static inline void DSPLIB_cholesky_solver_forwardSubstitution_ci(dataType *pL,
285  dataType *pY,
286  dataType *pB,
287  dataType *pDiv,
288  int32_t order,
289  int32_t colLstride,
290  uint8_t *pBlock)
291 {
292  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Entering function");
293 
294  __SE_TEMPLATE_v1 seDivReadParams;
295  __SA_TEMPLATE_v1 saDivStoreParams;
296  __SE_TEMPLATE_v1 seDiagReadParams;
297  __SA_TEMPLATE_v1 saDiagWriteParams;
298  seDiagReadParams = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + (2 * SE_PARAM_SIZE));
299  saDiagWriteParams = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + (3 * SE_PARAM_SIZE));
300  seDivReadParams = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + (4 * SE_PARAM_SIZE));
301  saDivStoreParams = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + (5 * SE_PARAM_SIZE));
302 
303  saDiagWriteParams.DIM1 = 1;
304 
305  typedef typename c7x::make_full_vector<dataType>::type vec;
306  int32_t eleCount = c7x::element_count_of<vec>::value;
307  int32_t nVec = DSPLIB_ceilingDiv(order, eleCount);
308 
309  /* Calculate reciprocals of Diagonal Elements */
310  __SE1_OPEN(pL, seDiagReadParams);
311  __SA1_OPEN(saDiagWriteParams);
312  int32_t row = 0;
313 
314  for (row = 0; row < order; row++) {
315  vec vecDiag = c7x::strm_eng<1, vec>::get_adv();
316 
317  __vpred predDiag = c7x::strm_agen<1, vec>::get_vpred();
318  vec *pStoreDiag = c7x::strm_agen<1, vec>::get_adv(pDiv);
319  __vstore_pred(predDiag, pStoreDiag, vecDiag);
320  }
321  __SE1_CLOSE();
322  __SA1_CLOSE();
323 
324  __SE0_OPEN(pDiv, seDivReadParams);
325  __SA0_OPEN(saDivStoreParams);
326  dataType TwoP0 = 2.0;
327  int32_t ii = 0;
328 
329  for (ii = 0; ii < nVec - 3; ii += 4) {
330  vec v1 = c7x::strm_eng<0, vec>::get_adv();
331  vec v2 = c7x::strm_eng<0, vec>::get_adv();
332  vec v3 = c7x::strm_eng<0, vec>::get_adv();
333  vec v4 = c7x::strm_eng<0, vec>::get_adv();
334 
335  vec yy1 = __recip(v1);
336  yy1 = yy1 * (TwoP0 - v1 * yy1);
337  yy1 = yy1 * (TwoP0 - v1 * yy1);
338 
339  vec yy2 = __recip(v2);
340  yy2 = yy2 * (TwoP0 - v2 * yy2);
341  yy2 = yy2 * (TwoP0 - v2 * yy2);
342 
343  vec yy3 = __recip(v3);
344  yy3 = yy3 * (TwoP0 - v3 * yy3);
345  yy3 = yy3 * (TwoP0 - v3 * yy3);
346 
347  vec yy4 = __recip(v4);
348  yy4 = yy4 * (TwoP0 - v4 * yy4);
349  yy4 = yy4 * (TwoP0 - v4 * yy4);
350 
351  __vpred predDiv1 = c7x::strm_agen<0, vec>::get_vpred();
352  vec *pStoreDiv1 = c7x::strm_agen<0, vec>::get_adv(pDiv);
353  __vstore_pred(predDiv1, pStoreDiv1, yy1);
354 
355  __vpred predDiv2 = c7x::strm_agen<0, vec>::get_vpred();
356  vec *pStoreDiv2 = c7x::strm_agen<0, vec>::get_adv(pDiv);
357  __vstore_pred(predDiv2, pStoreDiv2, yy2);
358 
359  __vpred predDiv3 = c7x::strm_agen<0, vec>::get_vpred();
360  vec *pStoreDiv3 = c7x::strm_agen<0, vec>::get_adv(pDiv);
361  __vstore_pred(predDiv3, pStoreDiv3, yy3);
362 
363  __vpred predDiv4 = c7x::strm_agen<0, vec>::get_vpred();
364  vec *pStoreDiv4 = c7x::strm_agen<0, vec>::get_adv(pDiv);
365  __vstore_pred(predDiv4, pStoreDiv4, yy4);
366  }
367 
368  for (; ii < nVec; ii++) {
369  vec v1 = c7x::strm_eng<0, vec>::get_adv();
370 
371  vec yy1 = __recip(v1);
372  yy1 = yy1 * (TwoP0 - v1 * yy1);
373  yy1 = yy1 * (TwoP0 - v1 * yy1);
374 
375  __vpred predDiv1 = c7x::strm_agen<0, vec>::get_vpred();
376  vec *pStoreDiv1 = c7x::strm_agen<0, vec>::get_adv(pDiv);
377  __vstore_pred(predDiv1, pStoreDiv1, yy1);
378  }
379 
380  __SE0_CLOSE();
381  __SA0_CLOSE();
382 
383  /*----------------------------------------------------------------------
384  Forward Substitution
385  -----------------------------------------------------------------------*/
386 
387  __SE_TEMPLATE_v1 seBlockParams;
388  __SE_TEMPLATE_v1 seReadYParams;
389  __SA_TEMPLATE_v1 saWriteYParams;
390  __SA_TEMPLATE_v1 sa1DReadParams;
391 
392  saWriteYParams = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + (3 * SE_PARAM_SIZE));
393  seReadYParams = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + (6 * SE_PARAM_SIZE));
394  seBlockParams = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + (7 * SE_PARAM_SIZE));
395  sa1DReadParams = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + (4 * SE_PARAM_SIZE));
396 
397  saWriteYParams.DIM1 = 1;
398  seReadYParams.DIM1 = 1;
399  seBlockParams.DIM1 = colLstride;
400 
401  dataType *pSE0 = pL;
402  dataType *pSA1 = pY;
403  dataType *pSA2 = pB;
404  dataType *pSA3 = pDiv;
405 
406  __SA1_OPEN(saWriteYParams);
407  __SA2_OPEN(sa1DReadParams);
408  __SA3_OPEN(sa1DReadParams);
409 
410  for (int32_t block = 0; block < nVec; block++) {
411  __vpred predB = c7x::strm_agen<2, vec>::get_vpred();
412  vec *pLoadB = c7x::strm_agen<2, vec>::get_adv(pSA2);
413  vec vecB = __vload_pred(predB, pLoadB);
414 
415  __vpred predDiv = c7x::strm_agen<3, vec>::get_vpred();
416  vec *pLoadDiv = c7x::strm_agen<3, vec>::get_adv(pSA3);
417  vec vecDiv = __vload_pred(predDiv, pLoadDiv);
418 
419  int32_t sumRows = block * eleCount;
420  int32_t totalRows = sumRows + eleCount;
421 
422  seBlockParams.ICNT1 = totalRows;
423  seReadYParams.ICNT1 = sumRows;
424 
425  __SE0_OPEN(pSE0, seBlockParams);
426  if (sumRows > 0) {
427  __SE1_OPEN(pY, seReadYParams);
428  }
429 
430  /* Calculate sum */
431  vec vecSum = (vec) 0;
432  vec vecSum1 = (vec) 0;
433  vec vecSum2 = (vec) 0;
434  vec vecSum3 = (vec) 0;
435  vec vecSum4 = (vec) 0;
436  int32_t vertical = 0;
437 
438  for (vertical = 0; vertical < sumRows - 3; vertical += 4) {
439  vec v1 = c7x::strm_eng<0, vec>::get_adv();
440  vec y1 = c7x::strm_eng<1, vec>::get_adv();
441  vecSum1 += v1 * y1;
442 
443  vec v2 = c7x::strm_eng<0, vec>::get_adv();
444  vec y2 = c7x::strm_eng<1, vec>::get_adv();
445  vecSum2 += v2 * y2;
446 
447  vec v3 = c7x::strm_eng<0, vec>::get_adv();
448  vec y3 = c7x::strm_eng<1, vec>::get_adv();
449  vecSum3 += v3 * y3;
450 
451  vec v4 = c7x::strm_eng<0, vec>::get_adv();
452  vec y4 = c7x::strm_eng<1, vec>::get_adv();
453  vecSum4 += v4 * y4;
454  }
455 
456  vecSum = vecSum1 + vecSum2 + vecSum3 + vecSum4;
457 
458  /* Calculate values */
459  dataType resultEle1;
460  for (vertical = 0; vertical < eleCount; vertical++) {
461  vec v1 = c7x::strm_eng<0, vec>::get_adv();
462  vec result1 = (vecB - vecSum) * vecDiv;
463  getElement(result1, vertical, &resultEle1);
464  vecSum += v1 * (resultEle1);
465  __vpred predYCalc = c7x::strm_agen<1, vec>::get_vpred();
466  vec *pStoreY = c7x::strm_agen<1, vec>::get_adv(pSA1);
467  __vstore_pred(predYCalc, pStoreY, (vec) resultEle1);
468  }
469 
470  pSE0 += eleCount;
471  __SE0_CLOSE();
472  __SE1_CLOSE();
473  }
474  __SA2_CLOSE();
475  __SA3_CLOSE();
476  __SA1_CLOSE();
477  DSPLIB_DEBUGPRINTFN(0, "%s\n", "Exiting function");
478 }

582  vec x4 = c7x::strm_eng<1, vec>::get_adv();
583  vecSum4 += v4 * x4;
584  }

595  getElement(result1, vecIndex--, &resultEle1);

597  dataType *pStoreX1 = c7x::strm_agen<1, dataType>::get_adv(pSA1);
598  *pStoreX1 = resultEle1;
599  }

614

616  saReverseParams.ICNT1 = 1;

621  __SE1_OPEN(pXLastElem, seReadXParams);
622  __SA2_OPEN(saReverseParams);
623  __SA3_OPEN(saReverseParams);
624

628  vec *pLoadY = c7x::strm_agen<2, vec>::get_adv(pY);
629  vec vecY = __vload_pred(predY, pLoadY);

632  vec *pLoadDiv = c7x::strm_agen<3, vec>::get_adv(pDiv);
633  vec vecDiv = __vload_pred(predDiv, pLoadDiv);
634 
635  vec vecSum = (vec) 0;
636  vec vecSum1 = (vec) 0;
637  vec vecSum2 = (vec) 0;
638  vec vecSum3 = (vec) 0;
639  vec vecSum4 = (vec) 0;
640  int32_t vertical = 0;
641

643  vec v1 = c7x::strm_eng<0, vec>::get_adv();
644  vec x1 = c7x::strm_eng<1, vec>::get_adv();
645  vecSum1 += v1 * x1;

648  vec x2 = c7x::strm_eng<1, vec>::get_adv();
649  vecSum2 += v2 * x2;

653  vecSum3 += v3 * x3;

656  vec x4 = c7x::strm_eng<1, vec>::get_adv();
657  vecSum4 += v4 * x4;
658  }
659 
660  vecSum = vecSum1 + vecSum2 + vecSum3 + vecSum4;

663  dataType resultEle;
664

677  __SE1_CLOSE();

679  __SA3_CLOSE();
