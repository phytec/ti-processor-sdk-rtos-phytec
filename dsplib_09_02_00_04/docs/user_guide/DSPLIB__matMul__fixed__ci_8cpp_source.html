362 template <typename dataTypeIn, typename dataTypeOut>
363 DSPLIB_STATUS DSPLIB_matMul_fixed_exec_ci(DSPLIB_kernelHandle handle,
364  void *restrict pIn0,
365  void *restrict pIn1,
366  void *restrict pOut)
367 {
368 #if ENABLE_PROFILE
369  uint64_t start = __TSC;
370  uint64_t overhead = __TSC - start;
371  uint64_t loopCycle = 0;
372  uint64_t accloopCycle = 0;
373  uint64_t count = 0;
374 #endif
375 
376  DSPLIB_STATUS status = DSPLIB_SUCCESS;
377 
378  DSPLIB_matMul_fixed_PrivArgs *pKerPrivArgs = (DSPLIB_matMul_fixed_PrivArgs *) handle;
379  uint8_t *pBlock = pKerPrivArgs->bufPblock;
380  int32_t MBlocks = pKerPrivArgs->MBlocks;
381  int32_t NBlocks = pKerPrivArgs->NBlocks;
382  int32_t KBlocks = pKerPrivArgs->KBlocks;
383  int32_t qs = pKerPrivArgs->qs;
384 
385  dataTypeIn *pIn0Local = (dataTypeIn *) pIn0;
386  dataTypeIn *pIn1Local = (dataTypeIn *) pIn1;
387  dataTypeIn *pOutLocal = (dataTypeIn *) pOut;
388 
389  __SE_TEMPLATE_v1 se0Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE0_PARAM_OFFSET);
390  __SE_TEMPLATE_v1 se1Params = *(__SE_TEMPLATE_v1 *) ((uint8_t *) pBlock + SE1_PARAM_OFFSET);
391  __SA_TEMPLATE_v1 sa0Params = *(__SA_TEMPLATE_v1 *) ((uint8_t *) pBlock + SA0_PARAM_OFFSET);
392 
393  __SE0_OPEN(pIn0Local, se0Params);
394  __SE1_OPEN(pIn1Local, se1Params);
395  __SA0_OPEN(sa0Params);
396 
397  c7x::uchar_vec vMask = setMask<dataTypeIn>();
398 
399  dataTypeOut shiftVec = (dataTypeOut) qs;
400  dataTypeOut zeroVec = (dataTypeOut) 0;
401  dataTypeOut outSum0 = zeroVec, outSum1 = zeroVec, outSum2 = zeroVec, outSum3 = zeroVec;
402  dataTypeOut outSum4 = zeroVec, outSum5 = zeroVec, outSum6 = zeroVec, outSum7 = zeroVec;
403  dataTypeOut outSum8 = zeroVec, outSum9 = zeroVec, outSum10 = zeroVec, outSum11 = zeroVec;
404  dataTypeOut outSum12 = zeroVec, outSum13 = zeroVec, outSum14 = zeroVec, outSum15 = zeroVec;
405  dataTypeOut out1, out2, out3, out4, out5, out6, out7, out8;
406 
407  for (int32_t l = 0; l < MBlocks * NBlocks; l++) {
408 #if ENABLE_PROFILE
409  start = __TSC;
410 #endif
411  for (int32_t p = 0; p < KBlocks; p++) {
412  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum0, outSum1);
413  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum2, outSum3);
414  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum4, outSum5);
415  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum6, outSum7);
416  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum8, outSum9);
417  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum10, outSum11);
418  vecMulAcc<dataTypeOut &, __SE_REG_0, __SE_REG_1_ADV>(outSum12, outSum13);
419  vecMulAcc<dataTypeOut &, __SE_REG_0_ADV, __SE_REG_1_ADV>(outSum14, outSum15);
420  }
421 #if ENABLE_PROFILE
422  loopCycle = __TSC - start;
423  accloopCycle += loopCycle;
424  printf("|INNER LOOP CNT : %d| %lu |\n", count++, loopCycle);
425 #endif
426 
427  vecShiftRight<dataTypeOut &>(outSum0, outSum1, shiftVec);
428  vecShiftRight<dataTypeOut &>(outSum2, outSum3, shiftVec);
