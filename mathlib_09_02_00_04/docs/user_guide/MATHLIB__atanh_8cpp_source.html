62 template <typename T>
63 static inline MATHLIB_STATUS
64 MATHLIB_atanh_log(__SE_TEMPLATE_v1 *se0Params, __SA_TEMPLATE_v1 *sa0Params, size_t length, T *pSrc0, T *pDst)
65 {
66  // variables
67  MATHLIB_STATUS status = MATHLIB_SUCCESS; // return function status
68  size_t numBlocks = 0; // compute loop's iteration count
69  size_t remNumBlocks = 0; // when numBlocks is not a multiple of SIMD width
70 
71  // derive c7x vector type from template typename
72  typedef typename c7x::make_full_vector<T>::type vec;
73 
74  // calculate compute loop's iteration counter
75  numBlocks = length / c7x::element_count_of<vec>::value;
76  remNumBlocks = length % c7x::element_count_of<vec>::value;
77  if (remNumBlocks) {
78  numBlocks++;
79  }
80 
81  if (status == MATHLIB_SUCCESS) {
82 
83  // open SE0, SE1, and SA0 for reading and writing operands
84  // MATHLIB_SE0SA0Open(se0Params, sa0Params, pDst);
85  MATHLIB_SE0SE1SA0Open(se0Params, sa0Params, pDst, pSrc0);
86 
87  /**********************************************************************/
88  /* Create and assign values for constants employed on pow computation */
89  /**********************************************************************/
90 
91  vec C1, C2, C3, C4, C5, pol_bound, half;
92  c7x::double_vec ln2;
93  c7x::uint_vec zero;
94  zero = (c7x::uint_vec) 0;
95 
96  ln2 = (c7x::double_vec) 0.693147180559945;
97  C1 = (vec) -0.2302894f;
98  C2 = (vec) 0.1908169f;
99  C3 = (vec) -0.2505905f;
100  C4 = (vec) 0.3333164f;
101  C5 = (vec) -0.5000002f;
102  pol_bound = (vec) 0.1f;
103  half = (vec) 0.5f;
104 
105  // compute loop to perform vector pow
106  for (size_t i = 0; i < numBlocks; i++) {
107  vec inVec = c7x::strm_eng<0, vec>::get_adv();
108 
109  /**********************************************************************/
110  /* Create variables employed on log computation */
111  /**********************************************************************/
112 
113  vec pol, r1, r2, r3, r4, outVec, inVecOriginal_abs;
114  c7x::double_vec inVecVals_odd, inVecVals_even, inVecVals_oddReciprocal, inVecVals_evenReciprocal,
115  inVecReciprocalApprox_8_15, inVecReciprocalApprox_0_7, inVecVals_8_15, inVecVals_0_7, rVals_0_7,
116  rVals_8_15, TVals_8_15, TVals_0_7, NVals_odd, NVals_even, NVals_0_7, NVals_8_15, pol_0_7, pol_8_15,
117  outVec_8_15, outVec_0_7;
118  c7x::uint_vec inVecReciprocal_32_63, inVecReciprocalClr_32_63, inVecReciprocalApprox_32_63, indexT,
119  upperBitsIndexT, lowerBitsIndexT;
120  c7x::int_vec N;
121 
122  /**********************************************************************/
123  /* Calculate Taylor series approximation for log */
124  /**********************************************************************/
125 
126  // Split vectors to compute r with double precision
127  inVecVals_odd = __high_float_to_double(inVec);
128  inVecVals_even = __low_float_to_double(inVec);
129  inVecVals_oddReciprocal = __recip(inVecVals_odd);
130  inVecVals_evenReciprocal = __recip(inVecVals_even);
131 
132  // Create floating point reciprocal approximation
133  // Upper 32 bits of all inVec reciprocal values
134  inVecReciprocal_32_63 = c7x::reinterpret<c7x::uint_vec>(__permute_odd_odd_int(
135  MATHLIB_vperm_data_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecVals_oddReciprocal),
136  c7x::reinterpret<c7x::uchar_vec>(inVecVals_evenReciprocal)));
137 
138  // Clear bits 0-16 inclusive
139  inVecReciprocalClr_32_63 = inVecReciprocal_32_63 & 0xFFFE0000;
140 
141  // Concatenate cleared bit reciprocal with zero bits
142  inVecReciprocalApprox_8_15 = c7x::reinterpret<c7x::double_vec>(__permute_high_high(
143  MATHLIB_vperm_data_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecReciprocalClr_32_63),
144  c7x::reinterpret<c7x::uchar_vec>(zero)));
145  inVecReciprocalApprox_0_7 = c7x::reinterpret<c7x::double_vec>(__permute_low_low(
146  MATHLIB_vperm_data_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecReciprocalClr_32_63),
147  c7x::reinterpret<c7x::uchar_vec>(zero)));
148 
149  // Split inVec into two vectors with double precision
150  inVecVals_0_7 = c7x::reinterpret<c7x::double_vec>(
151  __permute_low_low(MATHLIB_vperm_data_dp_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecVals_odd),
152  c7x::reinterpret<c7x::uchar_vec>(inVecVals_even)));
153  inVecVals_8_15 = c7x::reinterpret<c7x::double_vec>(
154  __permute_high_high(MATHLIB_vperm_data_dp_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecVals_odd),
155  c7x::reinterpret<c7x::uchar_vec>(inVecVals_even)));
156 
157  // Calculate r in double precision
158  rVals_0_7 = (inVecReciprocalApprox_0_7 * inVecVals_0_7) - 1.0;
159  rVals_8_15 = (inVecReciprocalApprox_8_15 * inVecVals_8_15) - 1.0;
160 
161  // Convert r to float, compute r to the power of 2, 3, 4
162  r1 = c7x::reinterpret<vec>(__permute_even_even_int(
163  MATHLIB_vperm_data_0_63, c7x::reinterpret<c7x::uchar_vec>(__double_to_float(rVals_8_15)),
164  c7x::reinterpret<c7x::uchar_vec>(__double_to_float(rVals_0_7))));
165  r2 = r1 * r1;
166  r3 = r1 * r2;
167  r4 = r2 * r2;
168 
169  // Compute Taylor series polynomial
170  pol = (C5 * r2) + ((C4 * r3) + ((((C2 * r1) + C3) + (C1 * r2)) * r4));
171 
172  /**********************************************************************/
173  /* Calculate N */
174  /**********************************************************************/
175 
176  // Upper 32 bits of all inVec reciprocal approximation values
177  inVecReciprocalApprox_32_63 = c7x::reinterpret<c7x::uint_vec>(__permute_odd_odd_int(
178  MATHLIB_vperm_data_0_63, c7x::reinterpret<c7x::uchar_vec>(inVecReciprocalApprox_8_15),
179  c7x::reinterpret<c7x::uchar_vec>(inVecReciprocalApprox_0_7)));
180 
181  N = c7x::convert<c7x::int_vec>(((inVecReciprocalApprox_32_63 << 1) >> 21) - 1023);
182 
183  // Covert N to double precision for later calculation with LUT values
184  NVals_odd = __high_int_to_double(N);
185  NVals_even = __low_int_to_double(N);
186  NVals_0_7 = c7x::reinterpret<c7x::double_vec>(__permute_low_low(MATHLIB_vperm_data_dp_interweave_0_63,
187  c7x::reinterpret<c7x::uchar_vec>(NVals_odd),
188  c7x::reinterpret<c7x::uchar_vec>(NVals_even)));
189  NVals_8_15 = c7x::reinterpret<c7x::double_vec>(
190  __permute_high_high(MATHLIB_vperm_data_dp_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(NVals_odd),
191  c7x::reinterpret<c7x::uchar_vec>(NVals_even)));
192 
193  /**********************************************************************/
194  /* Determine LUT values */
195  /**********************************************************************/
196 
197  // Calculate LUT index
198  indexT = (((inVecReciprocalApprox_32_63 << 12) >> 29) + MATHLIB_LOGTABLE_OFFSET);
199 
200  // Read from LUT and reconstruct double values split into two vectors
201  upperBitsIndexT = MATHLIB_LUTReadUpperBits(indexT);
202  lowerBitsIndexT = MATHLIB_LUTReadLowerBits(indexT);
203 
204  TVals_8_15 = c7x::reinterpret<c7x::double_vec>(
205  __permute_high_high(MATHLIB_vperm_data_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(upperBitsIndexT),
206  c7x::reinterpret<c7x::uchar_vec>(lowerBitsIndexT)));
207  TVals_0_7 = c7x::reinterpret<c7x::double_vec>(
208  __permute_low_low(MATHLIB_vperm_data_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(upperBitsIndexT),
209  c7x::reinterpret<c7x::uchar_vec>(lowerBitsIndexT)));
210 
211  // Calculate an adjusted T
212  TVals_8_15 = TVals_8_15 - (ln2 * NVals_8_15);
213  TVals_0_7 = TVals_0_7 - (ln2 * NVals_0_7);
214 
215  /**********************************************************************/
216  /* Calculate output with adjusted LUT and Taylor series values */
217  /**********************************************************************/
218 
219  // Split polynomial result into two vectors with double precision
220  pol_0_7 = c7x::reinterpret<c7x::double_vec>(__permute_low_low(
221  MATHLIB_vperm_data_dp_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(__high_float_to_double(pol)),
222  c7x::reinterpret<c7x::uchar_vec>(__low_float_to_double(pol))));
223  pol_8_15 = c7x::reinterpret<c7x::double_vec>(__permute_high_high(
224  MATHLIB_vperm_data_dp_interweave_0_63, c7x::reinterpret<c7x::uchar_vec>(__high_float_to_double(pol)),
225  c7x::reinterpret<c7x::uchar_vec>(__low_float_to_double(pol))));
226 
227  // Add LUT values and Taylor series values
228  outVec_0_7 = rVals_0_7 + TVals_0_7 + pol_0_7;
229  outVec_8_15 = rVals_8_15 + TVals_8_15 + pol_8_15;
230 
231  // Combine output vector into one floating point result
232  outVec = c7x::reinterpret<vec>(__permute_even_even_int(
233  MATHLIB_vperm_data_0_63, c7x::reinterpret<c7x::uchar_vec>(__double_to_float(outVec_8_15)),
234  c7x::reinterpret<c7x::uchar_vec>(__double_to_float(outVec_0_7))));
235 
236  outVec = outVec * half;
237 
238  /**********************************************************************/
239  /* Determine algorithm for specific input */
240  /**********************************************************************/
241 
242  // #if !defined(MATHLIB_FAST)
243  vec inVecOriginal = c7x::strm_eng<1, vec>::get_adv();
244  inVecOriginal_abs = __abs(inVecOriginal);
245  __vpred cmp_le_pol = __cmp_le_pred(inVecOriginal_abs, pol_bound);
246  outVec = __select(cmp_le_pol, inVec, outVec);
247  // #endif
248 
249  // Store atanh computation result in
250  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
251  vec *addr = c7x::strm_agen<0, vec>::get_adv(pDst);
252  __vstore_pred(tmp, addr, outVec);
253  }
254  MATHLIB_SE0SE1SA0Close();
255  }
256  return status;
257 }

285  /* Create and assign values for constants employed on pow computation */
286  /**********************************************************************/
287 
288  vec limit, zeroF, inf;
289 
290  limit = (vec) 1.0f;
291  zeroF = (vec) 0.0f;
292  inf = (vec) 0x7F800000u;
293 
294  // compute loop to perform vector pow
295  for (size_t i = 0; i < numBlocks; i++) {
296  vec inVec = c7x::strm_eng<0, vec>::get_adv();
297  vec inVecOriginal = c7x::strm_eng<1, vec>::get_adv();
298 
299  vec outVec, sign;
300 
301  vec inVecOriginal_abs = __abs(inVecOriginal);
302 
303  __vpred cmp_limit = __cmp_eq_pred(inVecOriginal_abs, limit);
304  outVec = __select(cmp_limit, inf, inVec);
305 
306  sign = (vec) 1.0;
307  __vpred cmp_lt_zero = __cmp_lt_pred(inVecOriginal, zeroF);
308  sign = __select(cmp_lt_zero, -sign, sign);
309 
310  outVec = outVec * sign;
311 
312  // Store exp computation result in
313  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
314  vec *addr = c7x::strm_agen<0, vec>::get_adv(pDst);
315  __vstore_pred(tmp, addr, outVec);
316  }
317  MATHLIB_SE0SE1SA0Close();
318  }
319  return status;
320 }

322 template <typename vecType> static inline vecType divspMod_atanh(vecType a, vecType b)
323 {
324  vecType res, r0, d0, d1, p0, p1, Two;
325 
326  Two = (vecType) 2.0;
327 
328  p0 = __recip(b);
329  d0 = p0 * b;
330  r0 = p0;
331 
332  p1 = Two - d0;
333  d1 = r0 * p1;
334 
335  res = a * d1;
336 
337  return res;
338 }

341 template <typename T> static inline void MATHLIB_atanh_vector(size_t length, T *pSrc, T *pDst)
342 {
343  // variables
344  size_t numBlocks = 0; // compute loop's iteration count
345  size_t remNumBlocks = 0; // when numBlocks is not a multiple of SIMD width
346 
347  // derive c7x vector type from template typename
348  typedef typename c7x::make_full_vector<float>::type vec;
349 
350  __SE_TEMPLATE_v1 se0Params = __gen_SE_TEMPLATE_v1();
351  __SA_TEMPLATE_v1 sa0Params = __gen_SA_TEMPLATE_v1();
352 
353  MATHLIB_SE0SA01DSequentialInit(&se0Params, &sa0Params, length, pSrc, pDst);
354 
355  // calculate compute loop's iteration counter
356  numBlocks = length / c7x::element_count_of<vec>::value;
357  remNumBlocks = length % c7x::element_count_of<vec>::value;
358  if (remNumBlocks) {
359  numBlocks++;
360  }
361 
362  // open SE0, SE1, and SA0 for reading and writing operands
363  MATHLIB_SE0SA0Open(&se0Params, &sa0Params, pSrc);
364 
365  /***********************************************************************/
366  /* Create and assign values for constants employed on acosh computation */
367  /***********************************************************************/
368 
369  vec pol_bound, c2, c4;
370  pol_bound = (vec) 0.1f;
371  c2 = (vec) 0.333327051f;
372  c4 = (vec) 0.202017226f;
373 
374  // compute loop to perform vector acosh
375  for (size_t i = 0; i < numBlocks; i++) {
376  vec inVec = c7x::strm_eng<0, vec>::get_adv();
377 
378  /**********************************************************************/
379  /* Create variables employed on acosh computation */
380  /**********************************************************************/
381 
382  vec x2, x4, pol, outVec, inVec_abs, temp1, temp2;
383 
384  inVec_abs = __abs(inVec);
385 
386  // #if !defined(MATHLIB_FAST)
387 
388  x2 = inVec * inVec;
389  x4 = x2 * x2;
390 
391  // use polynomial estimation for |x| <= 0.1
392  // polynomial to estimate atanh(x) for small inputs
393  pol = (c2 * x2) + (c4 * x4);
394  pol = (pol * inVec_abs) + inVec_abs;
395 
396  // #endif
397 
398  // calculate atanh(x) based on atanh(x) = 0.5*ln((1+ x) / (1- x))
399  temp1 = 1.0f + inVec_abs;
400  temp2 = 1.0f - inVec_abs;
401 
402  outVec = divspMod_atanh(temp1, temp2);
403 
404  // #if !defined(MATHLIB_FAST)
405 
406  __vpred cmp_le_pol = __cmp_le_pred(inVec_abs, pol_bound);
407  outVec = __select(cmp_le_pol, pol, outVec);
408 
409  // #endif
410 
411  __vpred tmp = c7x::strm_agen<0, vec>::get_vpred();
412  vec *addr = c7x::strm_agen<0, vec>::get_adv(pDst);
413  __vstore_pred(tmp, addr, outVec);
414  }
415 
416  MATHLIB_SE0SA0Close();
417 
418  MATHLIB_atanh_log(&se0Params, &sa0Params, length, pSrc, pDst);
419  MATHLIB_atanh_cond(&se0Params, &sa0Params, length, pSrc, pDst);
420 }
